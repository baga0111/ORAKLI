-- ამოცანა:
-- SHOP_ID-ის მიხედვით გამოიტანე:
-- count(*), sum(amount), avg(amount)
-- აჩვენე მხოლოდ shop-ები სადაც count >= 4

  shop_id,
  COUNT(*) AS orders_cnt,
  SUM(amount) AS total_amount,
  ROUND(AVG(amount), 2) AS avg_amount

ORDER BY SUM(amount) DESC;

-- TRAINING 6 — Non-Equi Join: amount grade + amount_above_min
--------------------------------------------------------------------------------
BEGIN EXECUTE IMMEDIATE 'DROP TABLE orders_t6 PURGE'; EXCEPTION WHEN OTHERS THEN NULL; END;

CREATE TABLE orders_t6 (

-- ამოცანა:
-- non-equi join: amount BETWEEN min_amount AND max_amount
-- გამოიტანე order_id, amount, grade, amount_above_min
-- დატოვე მხოლოდ grade 3,4,5
-- order by grade ASC, amount DESC

-- ამოხსნა:
SELECT
  o.order_id,
  o.amount,
  g.grade,
  (o.amount - g.min_amount) AS amount_above_min
FROM orders_t6 o
JOIN amount_grade_t6 g
  ON o.amount BETWEEN g.min_amount AND g.max_amount
WHERE g.grade IN (3,4,5)
ORDER BY g.grade ASC, o.amount DESC;

-- ამოცანა:
-- 1) დაადგინე ნორმალური ფორმა და ჩამოწერე FD-ები (კომენტარში)
-- 2) გააკეთე decomposition 4NF-მდე
-- 3) შექმენი PK/FK-ები და ჩატვირთე მონაცემები raw-დან

-- ამოხსნა (FD-ები/ლოგიკა კომენტარში):
-- FD მაგალითები:
-- customer_id  -> customer_name
-- product_code -> product_name
-- supplier_name -> supplier_phone
-- (raw ცხრილში განმეორებებია, ამიტომ 3NF/BCNF დარღვეულია ტრანზიტიული დამოკიდებულებებით)

-- 4NF-მდე გაყოფა:
-- customers_t10(customer_id PK, customer_name)
-- products_t10(product_code PK, product_name)
-- suppliers_t10(supplier_id PK, supplier_name UNIQUE, supplier_phone)
-- sales_t10(customer_id FK, product_code FK, supplier_id FK, sale_month, PK(customer_id, product_code, supplier_id, sale_month))
-- (თუ product-ს მრავალ supplier-თან ურთიერთობა გინდა ცალკე, დამატებით არის product_suppliers_t10)

CREATE TABLE customers_t10 (
  customer_id   NUMBER(6) PRIMARY KEY,
  customer_name VARCHAR2(50) NOT NULL
);

CREATE TABLE products_t10 (
  product_code VARCHAR2(10) PRIMARY KEY,
  product_name VARCHAR2(80) NOT NULL
);

CREATE TABLE suppliers_t10 (
  supplier_id   NUMBER PRIMARY KEY,
  supplier_name VARCHAR2(50) NOT NULL UNIQUE,
  supplier_phone VARCHAR2(20) NOT NULL
);

CREATE SEQUENCE seq_supplier_t10 START WITH 1;

CREATE TABLE sales_t10 (
  customer_id  NUMBER(6) NOT NULL,
  product_code VARCHAR2(10) NOT NULL,
  supplier_id  NUMBER NOT NULL,
  sale_month   VARCHAR2(10) NOT NULL,
  CONSTRAINT sales_t10_pk PRIMARY KEY (customer_id, product_code, supplier_id, sale_month),
  CONSTRAINT sales_t10_cust_fk FOREIGN KEY (customer_id) REFERENCES customers_t10(customer_id),
  CONSTRAINT sales_t10_prod_fk FOREIGN KEY (product_code) REFERENCES products_t10(product_code),
  CONSTRAINT sales_t10_supp_fk FOREIGN KEY (supplier_id) REFERENCES suppliers_t10(supplier_id)
);

-- სურვილისამებრ: Product-Supplier mapping (ბევრი-მრავალზე)
CREATE TABLE product_suppliers_t10 (
  product_code VARCHAR2(10) NOT NULL,
  supplier_id  NUMBER NOT NULL,
  CONSTRAINT product_suppliers_t10_pk PRIMARY KEY (product_code, supplier_id),
  CONSTRAINT ps_t10_prod_fk FOREIGN KEY (product_code) REFERENCES products_t10(product_code),
  CONSTRAINT ps_t10_supp_fk FOREIGN KEY (supplier_id) REFERENCES suppliers_t10(supplier_id)
);

-- მონაცემების ჩატვირთვა (ETL raw-დან):

INSERT INTO products_t10 (product_code, product_name)
SELECT DISTINCT product_code, product_name

-- suppliers: ვქმნით surrogate supplier_id-ს
INSERT INTO suppliers_t10 (supplier_id, supplier_name, supplier_phone)
SELECT seq_supplier_t10.NEXTVAL, s.supplier_name, s.supplier_phone
FROM (
  SELECT DISTINCT supplier_name, supplier_phone
  FROM sales_raw_t10
) s;

-- product_suppliers mapping
INSERT INTO product_suppliers_t10 (product_code, supplier_id)
SELECT DISTINCT r.product_code, sup.supplier_id
FROM sales_raw_t10 r
JOIN suppliers_t10 sup
  ON sup.supplier_name = r.supplier_name
 AND sup.supplier_phone = r.supplier_phone;

-- sales: თითო გაყიდვა/ჩანაწერი (customer, product, supplier, month)
INSERT INTO sales_t10 (customer_id, product_code, supplier_id, sale_month)
SELECT DISTINCT r.customer_id, r.product_code, sup.supplier_id, r.sale_month
FROM sales_raw_t10 r
JOIN suppliers_t10 sup
  ON sup.supplier_name = r.supplier_name
 AND sup.supplier_phone = r.supplier_phone;

COMMIT;
